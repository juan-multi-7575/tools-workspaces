from fastapi import FastAPI
from sse_starlette import EventSourceResponse
import asyncio
import json
app = FastAPI()
# Store active connections per workflow
workflow_subscribers: dict[str, list] = {}
@app.get("/workflows/{workflow_id}/status")
async def stream_status(workflow_id: str):
    async def event_generator():
        queue = asyncio.Queue()
        workflow_subscribers.setdefault(workflow_id, []).append(queue)
        
        try:
            while True:
                status_update = await queue.get()
                yield {
                    "event": "node_status",
                    "data": json.dumps(status_update)
                }
        finally:
            workflow_subscribers[workflow_id].remove(queue)
    
    return EventSourceResponse(event_generator())

import { useCallback, useEffect, useRef } from 'react';
import { useReactFlow, useNodesState } from '@xyflow/react';
function WorkflowCanvas({ workflowId }: { workflowId: string }) {
  const { setNodes } = useReactFlow();
  const eventSourceRef = useRef<EventSource | null>(null);
  useEffect(() => {
    const es = new EventSource(`/workflows/${workflowId}/status`);
    eventSourceRef.current = es;
    es.addEventListener('node_status', (event) => {
      const { node_id, status } = JSON.parse(event.data);
      
      setNodes((nodes) =>
        nodes.map((node) =>
          node.id === node_id
            ? {
                ...node,
                data: { ...node.data, status },
                className: `node-${status}`, // CSS class for visual feedback
              }
            : node
        )
      );
    });
    return () => es.close();
  }, [workflowId, setNodes]);
  return <ReactFlow />;
}

import { create } from 'zustand';
import { persist } from 'zustand/middleware';
import { Node, Edge, Viewport } from '@xyflow/react';
interface WorkflowState {
  nodes: Node[];
  edges: Edge[];
  viewport: Viewport;
  setNodes: (nodes: Node[]) => void;
  setEdges: (edges: Edge[]) => void;
  setViewport: (viewport: Viewport) => void;
}
export const useWorkflowStore = create<WorkflowState>()(
  persist(
    (set) => ({
      nodes: [],
      edges: [],
      viewport: { x: 0, y: 0, zoom: 1 },
      
      setNodes: (nodes) => set({ nodes }),
      setEdges: (edges) => set({ edges }),
      setViewport: (viewport) => set({ viewport }),

    {
      name: 'workflow-state',
      storage: createJSONStorage(() => localStorage), // or indexedDB for large graphs
    }
  )
);

  return (
    <ReactFlow
      nodes={nodes}
      edges={edges}
      onNodesChange={onNodesChange}
      onEdgesChange={(changes) => {
        const updatedEdges = applyEdgeChanges(changes, edges);
        setEdges(updatedEdges);
      }}
      defaultViewport={viewport}
      onMoveEnd={onMoveEnd}
    >
      <Background />
      <Controls />
    </ReactFlow>
  );
}
